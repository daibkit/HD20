/**
 * HDX20 — cầu nối giữa website và Google Drive (Bảo tàng ký ức).
 * Ảnh được lưu trong thư mục "HDX20 - Bảo tàng ký ức" ở Drive của BẠN, mỗi khóa một thư mục con (K48, K49…).
 * Chỉ tài khoản admin của website mới tải lên / xoá / liệt kê được (kiểm tra qua Supabase).
 * Hai giá trị bên dưới được admin tự điền khi bấm "Sao chép mã"; nếu dán tay thì thay vào.
 */
const SUPABASE_URL = '__SUPABASE_URL__';
const SUPABASE_KEY = '__SUPABASE_KEY__';
const ROOT_NAME = 'HDX20 - Bảo tàng ký ức';

// Trả về '' nếu token thuộc admin, ngược lại trả về lý do (hiện trong thông báo lỗi).
function adminProblem_(token) {
  if (!SUPABASE_URL || SUPABASE_URL.indexOf('__') >= 0 || !SUPABASE_KEY || SUPABASE_KEY.indexOf('__') >= 0)
    return 'mã Apps Script chưa có địa chỉ/khóa Supabase (hai dòng SUPABASE_URL và SUPABASE_KEY ở đầu mã) — hãy bấm “Sao chép mã” trong trang quản trị và dán lại';
  if (!token || typeof token !== 'string') return 'thiếu phiên đăng nhập, hãy đăng nhập lại ở trang quản trị';
  try {
    let base = String(SUPABASE_URL).trim().replace(/\/+$/, '');
    if (/^[a-z0-9]{20}$/.test(base)) base = 'https://' + base + '.supabase.co'; // cho phép chỉ dán mã dự án
    if (!/^https:\/\//.test(base)) return 'SUPABASE_URL phải có dạng https://xxxx.supabase.co';
    const h = {apikey: SUPABASE_KEY, Authorization: 'Bearer ' + token, 'Content-Type': 'application/json'};
    const r = UrlFetchApp.fetch(base + '/rest/v1/rpc/is_admin', {method: 'post', headers: h, payload: '{}', muteHttpExceptions: true});
    const code = r.getResponseCode(), body = String(r.getContentText()).trim();
    if (code === 200 && body === 'true') return '';
    if (code === 200) return 'tài khoản đăng nhập không nằm trong danh sách admin';
    if (code === 401) return 'phiên đăng nhập hết hạn, hãy tải lại trang quản trị và đăng nhập lại';
    return 'Supabase trả về mã ' + code + ' (kiểm tra SUPABASE_URL / SUPABASE_KEY)';
  } catch (e) { return 'Apps Script chưa được phép gọi ra ngoài hoặc địa chỉ Supabase sai: ' + String(e && e.message || e).slice(0, 120); }
}
// Nhớ kết quả kiểm tra admin 5 phút để tải nhiều phần không phải hỏi Supabase mỗi lần.
function adminCached_(token) {
  const c = CacheService.getScriptCache();
  const key = 'adm' + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(token)));
  if (c.get(key) === '1') return '';
  const why = adminProblem_(token);
  if (!why) c.put(key, '1', 300);
  return why;
}
const OK_MIME_ = /^(image\/(jpeg|png|webp|gif|heic|heif|avif)|video\/(mp4|quicktime|webm|x-m4v|3gpp|x-matroska|mpeg|ogg))$/;
function root_() {
  const p = PropertiesService.getScriptProperties(); let id = p.getProperty('ROOT_ID');
  if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} }
  const it = DriveApp.getFoldersByName(ROOT_NAME);
  const f = it.hasNext() ? it.next() : DriveApp.createFolder(ROOT_NAME);
  p.setProperty('ROOT_ID', f.getId()); return f;
}
function folder_(k) {
  if (k === 'VIDEO') k = 'Video giới thiệu'; // video ở trang chủ
  else if (!/^K[0-9]{1,3}$/.test(k)) throw new Error('Tên thư mục không hợp lệ');
  const r = root_(), it = r.getFoldersByName(k);
  return it.hasNext() ? it.next() : r.createFolder(k);
}
function out_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function doGet() { return out_({ok: true, name: 'HDX20 Drive', note: 'Dùng POST từ trang quản trị.'}); }

function doPost(e) {
  try {
    const q = JSON.parse(e.postData.contents);
    const why = adminCached_(q.token);
    if (why) return out_({ok: false, error: 'Không có quyền: ' + why + '.'});
    if (q.action === 'ping') return out_({ok: true, folder: root_().getName()});
    if (q.action === 'start') { // mở phiên tải lên theo từng phần (cho ảnh/video dung lượng lớn)
      if (!OK_MIME_.test(q.mime || '')) return out_({ok: false, error: 'Chỉ nhận ảnh (JPG/PNG/WebP/GIF/HEIC) và video (MP4/MOV/WebM…).'});
      const size = Number(q.size); if (!(size > 0)) return out_({ok: false, error: 'Tệp rỗng.'});
      const name = String(q.name || 'tep').replace(/[^\w.\- ]/g, '_').slice(0, 120);
      const r = UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&fields=id', {
        method: 'post', contentType: 'application/json; charset=UTF-8', muteHttpExceptions: true,
        headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken(), 'X-Upload-Content-Type': q.mime, 'X-Upload-Content-Length': String(size)},
        payload: JSON.stringify({name: name, parents: [folder_(q.k).getId()]})});
      const hs = r.getAllHeaders(); let loc = '';
      for (const h in hs) if (h.toLowerCase() === 'location') loc = String(hs[h]);
      if (r.getResponseCode() !== 200 || !loc) return out_({ok: false, error: 'Google Drive từ chối mở phiên tải lên (' + r.getResponseCode() + '): ' + String(r.getContentText()).slice(0, 160)});
      return out_({ok: true, session: loc});
    }
    if (q.action === 'chunk') {
      const sess = String(q.session || '');
      if (sess.indexOf('https://www.googleapis.com/upload/drive/v3/files?') !== 0) return out_({ok: false, error: 'Phiên tải lên không hợp lệ.'});
      const bytes = Utilities.base64Decode(q.data || ''), start = Number(q.start), total = Number(q.total);
      if (!bytes.length || !(total > 0) || !(start >= 0)) return out_({ok: false, error: 'Dữ liệu không hợp lệ.'});
      const end = start + bytes.length - 1;
      const r = UrlFetchApp.fetch(sess, {method: 'put', contentType: 'application/octet-stream', muteHttpExceptions: true, payload: bytes,
        headers: {'Content-Range': 'bytes ' + start + '-' + end + '/' + total}});
      const code = r.getResponseCode();
      if (code === 308) return out_({ok: true, done: false});
      if (code === 200 || code === 201) {
        const id = JSON.parse(r.getContentText()).id, f = DriveApp.getFileById(id);
        f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        return out_({ok: true, done: true, id: id, name: f.getName()});
      }
      return out_({ok: false, error: 'Google Drive báo lỗi ' + code + ': ' + String(r.getContentText()).slice(0, 160)});
    }
    if (q.action === 'init') { // tạo sẵn đủ thư mục K48…K67 + “Video giới thiệu” (có rồi thì giữ nguyên)
      const names = [], made = [];
      for (let n = 48; n <= 67; n++) names.push('K' + n);
      names.push('VIDEO');
      names.forEach(function (k) { const had = k === 'VIDEO' ? root_().getFoldersByName('Video giới thiệu').hasNext() : root_().getFoldersByName(k).hasNext(); folder_(k); if (!had) made.push(k); });
      return out_({ok: true, total: names.length, created: made.length, root: root_().getName()});
    }
    if (q.action === 'list') {
      const it = folder_(q.k).getFiles(), files = [];
      while (it.hasNext()) {
        const f = it.next(); if (!/^(image|video)\//.test(f.getMimeType())) continue;
        if (f.getSharingAccess() !== DriveApp.Access.ANYONE_WITH_LINK && f.getSharingAccess() !== DriveApp.Access.ANYONE) f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        files.push({id: f.getId(), name: f.getName(), mime: f.getMimeType(), t: f.getDateCreated().getTime()});
      }
      files.sort(function (a, b) { return a.t - b.t; });
      return out_({ok: true, files: files});
    }
    if (q.action === 'delete') {
      if (!/^[\w-]{10,100}$/.test(q.id || '')) return out_({ok: false, error: 'Mã tệp không hợp lệ.'});
      const f = DriveApp.getFileById(q.id), r = root_(); let inside = false, ps = f.getParents();
      while (ps.hasNext()) { const p = ps.next(), pp = p.getParents(); if (pp.hasNext() && pp.next().getId() === r.getId()) inside = true; }
      if (!inside) return out_({ok: false, error: 'Tệp không nằm trong thư mục Bảo tàng ký ức.'});
      f.setTrashed(true); return out_({ok: true});
    }
    return out_({ok: false, error: 'Thao tác không hợp lệ.'});
  } catch (err) { return out_({ok: false, error: String(err && err.message || err)}); }
}
