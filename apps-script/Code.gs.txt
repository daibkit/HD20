/**
 * HDX20 — cầu nối giữa website và Google Drive (Bảo tàng ký ức).
 * Ảnh được lưu trong thư mục "HDX20 - Bảo tàng ký ức" ở Drive của BẠN, mỗi khóa một thư mục con (K48, K49…).
 * Chỉ tài khoản admin của website mới tải lên / xoá / liệt kê được (kiểm tra qua Supabase).
 * Hai giá trị bên dưới được admin tự điền khi bấm "Sao chép mã"; nếu dán tay thì thay vào.
 */
const SUPABASE_URL = '__SUPABASE_URL__';
const SUPABASE_KEY = '__SUPABASE_KEY__';
const ROOT_NAME = 'HDX20 - Bảo tàng ký ức';

// CHẠY HÀM NÀY MỘT LẦN trong trình soạn thảo (chọn "authorize" → ▶ Chạy → Cho phép) để cấp quyền Drive + kết nối ngoài.
// Nếu không, website sẽ báo "Truy cập bị từ chối: DriveApp". Sau đó Triển khai → Phiên bản mới.
function authorize() {
  DriveApp.getRootFolder();
  UrlFetchApp.fetch('https://www.googleapis.com/discovery/v1/apis', {muteHttpExceptions: true});
  const f = root_();
  sheet_(); // tạo sẵn Google Sheet đăng ký (cấp quyền Google Sheets)
  const why = share_(f);
  Logger.log(why ? 'KHÔNG bật được chia sẻ công khai: ' + why + ' — có thể quản trị viên Google Workspace chặn chia sẻ ra ngoài, hoặc thư mục không thuộc tài khoản này (ảnh vẫn tải lên được, chỉ có thể không hiện công khai).' : 'Thư mục gốc đã mở “bất kỳ ai có liên kết” (giữ nguyên quyền hiện có).');
  Logger.log('OK — đã cấp quyền. Thư mục gốc: ' + f.getUrl() + ' · Sheet đăng ký: ' + sheet_().getUrl());
}

// Trả về '' nếu token thuộc admin, ngược lại trả về lý do (hiện trong thông báo lỗi).
function adminProblem_(token) {
  if (!SUPABASE_URL || SUPABASE_URL.indexOf('__') >= 0 || !SUPABASE_KEY || SUPABASE_KEY.indexOf('__') >= 0)
    return 'mã Apps Script chưa có địa chỉ/khóa Supabase (hai dòng SUPABASE_URL và SUPABASE_KEY ở đầu mã) — hãy bấm “Sao chép mã” trong trang quản trị và dán lại';
  if (!token || typeof token !== 'string') return 'thiếu phiên đăng nhập, hãy đăng nhập lại ở trang quản trị';
  try {
    let base = String(SUPABASE_URL).trim().replace(/\/+$/, '');
    if (/^[a-z0-9]{20}$/.test(base)) base = 'https://' + base + '.supabase.co'; // cho phép chỉ dán mã dự án
    if (!/^https:\/\//.test(base)) return 'SUPABASE_URL phải có dạng https://xxxx.supabase.co';
    const h = {apikey: SUPABASE_KEY, Authorization: 'Bearer ' + token, 'Content-Type': 'application/json'};
    const r = UrlFetchApp.fetch(base + '/rest/v1/rpc/is_admin', {method: 'post', headers: h, payload: '{}', muteHttpExceptions: true});
    const code = r.getResponseCode(), body = String(r.getContentText()).trim();
    if (code === 200 && body === 'true') return '';
    if (code === 200) return 'tài khoản đăng nhập không nằm trong danh sách admin';
    if (code === 401) return 'phiên đăng nhập hết hạn, hãy tải lại trang quản trị và đăng nhập lại';
    return 'Supabase trả về mã ' + code + ' (kiểm tra SUPABASE_URL / SUPABASE_KEY)';
  } catch (e) { return 'Apps Script chưa được phép gọi ra ngoài hoặc địa chỉ Supabase sai: ' + String(e && e.message || e).slice(0, 120); }
}
// Nhớ kết quả kiểm tra admin 5 phút để tải nhiều phần không phải hỏi Supabase mỗi lần.
function adminCached_(token) {
  const c = CacheService.getScriptCache();
  const key = 'adm' + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(token)));
  if (c.get(key) === '1') return '';
  const why = adminProblem_(token);
  if (!why) c.put(key, '1', 300);
  return why;
}
const OK_MIME_ = /^(image\/(jpeg|png|webp|gif|heic|heif|avif)|video\/(mp4|quicktime|webm|x-m4v|3gpp|x-matroska|mpeg|ogg))$/;
// CTX_: ghi lại đang làm gì / với thư mục nào, để khi lỗi quyền thì báo rõ ràng.
var CTX_ = {step: '', folder: '', id: ''};
// Bật “ai có liên kết đều xem được” NẾU CẦN. Đã mở sẵn (kể cả cho phép sửa) thì giữ nguyên, không đụng vào; không bao giờ ném lỗi.
// Trả về '' nếu ổn, hoặc lý do nếu không bật được.
function share_(item) {
  try { const a = item.getSharingAccess(); if (a === DriveApp.Access.ANYONE || a === DriveApp.Access.ANYONE_WITH_LINK) return ''; } catch (e) {}
  try { item.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return ''; }
  catch (e) { return String(e && e.message || e); }
}
function ctx_(step, folder, id) { CTX_.step = step || ''; if (folder !== undefined) CTX_.folder = folder; if (id !== undefined) CTX_.id = id || ''; }
function root_() {
  ctx_('mở thư mục gốc “' + ROOT_NAME + '”', ROOT_NAME, '');
  const p = PropertiesService.getScriptProperties(); let id = p.getProperty('ROOT_ID');
  if (id) { try { const f0 = DriveApp.getFolderById(id); CTX_.id = id; return f0; } catch (e) { Logger.log('Không mở được thư mục gốc theo ID đã lưu (' + id + '): ' + e.message); } }
  const it = DriveApp.getFoldersByName(ROOT_NAME);
  const f = it.hasNext() ? it.next() : DriveApp.createFolder(ROOT_NAME);
  p.setProperty('ROOT_ID', f.getId());
  share_(f);
  return f;
}
function folder_(k) {
  if (k === 'VIDEO') k = 'Video giới thiệu'; // video ở trang chủ
  else if (!/^K[0-9]{1,3}$/.test(k)) throw new Error('Tên thư mục không hợp lệ');
  const r = root_(); ctx_('mở/tạo thư mục con “' + k + '” trong “' + ROOT_NAME + '”', k, '');
  const it = r.getFoldersByName(k), f = it.hasNext() ? it.next() : r.createFolder(k);
  CTX_.id = f.getId(); return f;
}
function out_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }


// ---------- Đăng ký tham dự: ghi vào Google Sheet (nằm riêng ở Drive của bạn, KHÔNG nằm trong thư mục chia sẻ công khai) ----------
const SHEET_NAME = 'HDX20 - Đăng ký tham dự';
const SHEET_HEAD = ['Thời gian', 'Họ và tên', 'Khóa', 'Số điện thoại', 'Tiệc tối', 'Lời nhắn', 'Đã liên hệ', 'Ghi chú BTC'];
function sheet_() {
  const p = PropertiesService.getScriptProperties(), id = p.getProperty('SHEET_ID');
  if (id) { try { return SpreadsheetApp.openById(id); } catch (e) {} }
  const ss = SpreadsheetApp.create(SHEET_NAME), sh = ss.getSheets()[0];
  sh.setName('Đăng ký'); sh.getRange('A:H').setNumberFormat('@'); // định dạng văn bản: số điện thoại không mất số 0, không bị hiểu thành công thức
  sh.appendRow(SHEET_HEAD); sh.getRange(1, 1, 1, SHEET_HEAD.length).setFontWeight('bold').setBackground('#E6F3FB'); sh.setFrozenRows(1);
  sh.setColumnWidths(1, 1, 140); sh.setColumnWidths(2, 1, 190); sh.setColumnWidths(4, 1, 120); sh.setColumnWidths(6, 1, 320);
  p.setProperty('SHEET_ID', ss.getId()); return ss;
}
function cell_(v, n) { return String(v == null ? '' : v).replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '').trim().slice(0, n); }
function now_() { return Utilities.formatDate(new Date(), 'Asia/Ho_Chi_Minh', 'dd/MM/yyyy HH:mm:ss'); }
// Công khai: form trên website gọi để thêm một dòng (bản chính vẫn nằm trong Supabase).
function register_(q) {
  const name = cell_(q.name, 80), phone = cell_(q.phone, 20);
  if (!name) return {ok: false, error: 'Thiếu họ và tên.'};
  if (!/^[0-9+\s.]{9,20}$/.test(phone)) return {ok: false, error: 'Số điện thoại chưa hợp lệ.'};
  const cache = CacheService.getScriptCache(), dk = 'reg' + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, name + '|' + phone));
  if (cache.get(dk)) return {ok: true, duplicate: true};
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const minute = 'rl' + Math.floor(Date.now() / 60000), n = Number(cache.get(minute) || 0);
    if (n >= 40) return {ok: false, error: 'Quá nhiều đăng ký trong một phút.'};
    cache.put(minute, String(n + 1), 120);
    sheet_().getSheets()[0].appendRow([now_(), name, cell_(q.k, 20), phone, cell_(q.dinner, 20), cell_(q.note, 600), '', '']);
    cache.put(dk, '1', 3600);
  } finally { lock.releaseLock(); }
  return {ok: true};
}
// Admin: ghi lại TOÀN BỘ danh sách (lấy từ Supabase) vào sheet — chạy lại bao nhiêu lần cũng không bị trùng.
function sheetSync_(q) {
  const rows = (Array.isArray(q.rows) ? q.rows : []).slice(0, 20000).map(function (r) {
    r = Array.isArray(r) ? r : []; const lim = [30, 80, 20, 20, 20, 600, 10, 600];
    return lim.map(function (n, i) { return cell_(r[i], n); });
  });
  const lock = LockService.getScriptLock(); lock.waitLock(30000);
  try {
    const ss = sheet_(), sh = ss.getSheets()[0], last = sh.getLastRow();
    if (last > 1) sh.getRange(2, 1, last - 1, SHEET_HEAD.length).clearContent();
    if (rows.length) sh.getRange(2, 1, rows.length, SHEET_HEAD.length).setNumberFormat('@').setValues(rows);
    return {ok: true, count: rows.length, url: ss.getUrl()};
  } finally { lock.releaseLock(); }
}

function doGet() { return out_({ok: true, name: 'HDX20 Drive', note: 'Dùng POST từ trang quản trị.'}); }

function doPost(e) {
  try {
    const q = JSON.parse(e.postData.contents);
    if (q.action === 'register') return out_(register_(q)); // công khai: ai cũng gửi được form đăng ký
    const why = adminCached_(q.token);
    if (why) return out_({ok: false, error: 'Không có quyền: ' + why + '.'});
    if (q.action === 'check') { // kiểm tra quyền từng thư mục, báo rõ thư mục nào lỗi
      const res = [];
      const test = function (label, fn) { try { const v = fn(); res.push({name: label, ok: true, info: v || ''}); } catch (e) { res.push({name: label, ok: false, info: String(e && e.message || e)}); } };
      let r = null; test('Thư mục gốc “' + ROOT_NAME + '”', function () { r = root_(); return r.getUrl() + ' · chủ sở hữu: ' + (function () { try { return r.getOwner().getEmail(); } catch (e) { return '?'; } })(); });
      if (r) {
        const names = []; for (let n = 48; n <= 67; n++) names.push('K' + n); names.push('Video giới thiệu');
        names.forEach(function (k) { test('Thư mục “' + k + '”', function () { const it = r.getFoldersByName(k); if (!it.hasNext()) return 'chưa có (sẽ tự tạo khi tải lên)'; const f = it.next(); f.getFiles().hasNext(); return 'đọc được; ID ' + f.getId(); }); });
        test('Ghi thử vào thư mục gốc', function () { const t = r.createFile('_kiem-tra-quyen.txt', 'ok'); t.setTrashed(true); return 'ghi + xoá thử được'; });
      }
      test('Google Sheet đăng ký', function () { const ss = sheet_(); return ss.getUrl(); });
      const bad = res.filter(function (x) { return !x.ok; });
      Logger.log(JSON.stringify(res));
      return out_({ok: true, results: res, failed: bad.length});
    }
    if (q.action === 'ping') { const sid = PropertiesService.getScriptProperties().getProperty('SHEET_ID'); let su = ''; if (sid) { try { su = SpreadsheetApp.openById(sid).getUrl(); } catch (e) {} } return out_({ok: true, folder: root_().getName(), sheet: su}); }
    if (q.action === 'sheetsync') return out_(sheetSync_(q));
    if (q.action === 'start') { // mở phiên tải lên theo từng phần (cho ảnh/video dung lượng lớn)
      if (!OK_MIME_.test(q.mime || '')) return out_({ok: false, error: 'Chỉ nhận ảnh (JPG/PNG/WebP/GIF/HEIC) và video (MP4/MOV/WebM…).'});
      const size = Number(q.size); if (!(size > 0)) return out_({ok: false, error: 'Tệp rỗng.'});
      const name = String(q.name || 'tep').replace(/[^\w.\- ]/g, '_').slice(0, 120);
      const r = UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&fields=id', {
        method: 'post', contentType: 'application/json; charset=UTF-8', muteHttpExceptions: true,
        headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken(), 'X-Upload-Content-Type': q.mime, 'X-Upload-Content-Length': String(size)},
        payload: JSON.stringify({name: name, parents: [folder_(q.k).getId()]})});
      const hs = r.getAllHeaders(); let loc = '';
      for (const h in hs) if (h.toLowerCase() === 'location') loc = String(hs[h]);
      if (r.getResponseCode() !== 200 || !loc) return out_({ok: false, error: 'Google Drive từ chối mở phiên tải lên (' + r.getResponseCode() + '): ' + String(r.getContentText()).slice(0, 160)});
      return out_({ok: true, session: loc});
    }
    if (q.action === 'chunk') {
      const sess = String(q.session || '');
      if (sess.indexOf('https://www.googleapis.com/upload/drive/v3/files?') !== 0) return out_({ok: false, error: 'Phiên tải lên không hợp lệ.'});
      const bytes = Utilities.base64Decode(q.data || ''), start = Number(q.start), total = Number(q.total);
      if (!bytes.length || !(total > 0) || !(start >= 0)) return out_({ok: false, error: 'Dữ liệu không hợp lệ.'});
      const end = start + bytes.length - 1;
      const r = UrlFetchApp.fetch(sess, {method: 'put', contentType: 'application/octet-stream', muteHttpExceptions: true, payload: bytes,
        headers: {'Content-Range': 'bytes ' + start + '-' + end + '/' + total}});
      const code = r.getResponseCode();
      if (code === 308) return out_({ok: true, done: false});
      if (code === 200 || code === 201) {
        const id = JSON.parse(r.getContentText()).id;
        let where = '', url = '', name = '', shared = true, why = '';
        try { const f = DriveApp.getFileById(id); name = f.getName(); const par = f.getParents().next(); where = par.getName(); url = par.getUrl(); } catch (e) { why = String(e && e.message || e); }
        const sw = share_(DriveApp.getFileById(id)); if (sw) { shared = false; why = sw; }
        return out_({ok: true, done: true, id: id, name: name, folder: where, folderUrl: url, shared: shared, warn: why});
      }
      return out_({ok: false, error: 'Google Drive báo lỗi ' + code + ': ' + String(r.getContentText()).slice(0, 160)});
    }
    if (q.action === 'init') { // tạo sẵn đủ thư mục K48…K67 + “Video giới thiệu” (có rồi thì giữ nguyên)
      const names = [], made = [];
      for (let n = 48; n <= 67; n++) names.push('K' + n);
      names.push('VIDEO');
      names.forEach(function (k) { const had = k === 'VIDEO' ? root_().getFoldersByName('Video giới thiệu').hasNext() : root_().getFoldersByName(k).hasNext(); folder_(k); if (!had) made.push(k); });
      return out_({ok: true, total: names.length, created: made.length, root: root_().getName()});
    }
    if (q.action === 'list') { // có phân trang: offset/limit để trang quản trị hiện được tiến trình
      const it = folder_(q.k).getFiles(), all = [];
      while (it.hasNext()) {
        const f = it.next(); if (!/^(image|video)\//.test(f.getMimeType())) continue;
        all.push({f: f, t: f.getDateCreated().getTime()});
      }
      all.sort(function (a, b) { return a.t - b.t; });
      const off = Math.max(0, Number(q.offset) || 0), lim = q.limit ? Math.min(60, Math.max(1, Number(q.limit))) : all.length;
      const files = all.slice(off, off + lim).map(function (x) {
        const f = x.f;
        share_(f);
        return {id: f.getId(), name: f.getName(), mime: f.getMimeType(), t: x.t};
      });
      return out_({ok: true, files: files, total: all.length, next: off + files.length});
    }
    if (q.action === 'delete') {
      if (!/^[\w-]{10,100}$/.test(q.id || '')) return out_({ok: false, error: 'Mã tệp không hợp lệ.'});
      const f = DriveApp.getFileById(q.id), r = root_(); let inside = false, ps = f.getParents();
      while (ps.hasNext()) { const p = ps.next(), pp = p.getParents(); if (pp.hasNext() && pp.next().getId() === r.getId()) inside = true; }
      if (!inside) return out_({ok: false, error: 'Tệp không nằm trong thư mục Bảo tàng ký ức.'});
      f.setTrashed(true); return out_({ok: true});
    }
    return out_({ok: false, error: 'Thao tác không hợp lệ.'});
  } catch (err) {
    const msg = String(err && err.message || err);
    const where = (CTX_.step ? 'Bước: ' + CTX_.step : '') + (CTX_.folder ? ' · Thư mục: “' + CTX_.folder + '”' : '') + (CTX_.id ? ' (ID ' + CTX_.id + ')' : '');
    const perm = /access|denied|permission|quyền|authoriz|scope|not have/i.test(msg);
    Logger.log('LỖI ' + (perm ? '[QUYỀN] ' : '') + msg + ' — ' + where);
    return out_({ok: false, error: msg + (where ? ' — ' + where : '') + (perm ? ' — Tài khoản chạy Apps Script chưa có quyền với thư mục/tệp trên. Hãy mở Drive bằng ĐÚNG tài khoản đã triển khai Apps Script, kiểm tra thư mục này có nằm trong Drive của tài khoản đó không (hoặc được chia sẻ quyền “Người chỉnh sửa”), rồi chạy lại authorize().' : '')});
  }
}
